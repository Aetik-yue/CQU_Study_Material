\section{ALU代码 (alu.v)}

\begin{lstlisting}[language=Verilog]
module alu (
    input  [31:0] A,
    input  [31:0] B,
    input  [2:0]  F,
    output reg [31:0] Result
);

    always @(*) begin
        case (F)
            3'b000: Result = A + B;
            3'b001: Result = A - B;
            3'b010: Result = A & B;
            3'b011: Result = A | B;
            3'b100: Result = ~A;
            3'b101: Result = ($signed(A) < $signed(B)) ? 32'd1 : 32'd0;
            default: Result = 32'd0;
        endcase
    end

endmodule
\end{lstlisting}

\section{4级流水线32bit全加器代码 (liushuixian\_4ji.v)}

\begin{lstlisting}[language=Verilog]
module liushuixian_4ji (
    input [31:0] cin_a, cin_b,
    input cin, clk,
    input stall, flush,
    output reg cout,
    output reg [31:0] sum
);
    reg [7:0] sum_s1, sum_s2, sum_s3, sum_s4;
    reg [23:0] rem_a1, rem_b1;
    reg [15:0] rem_a2, rem_b2;
    reg [7:0]  rem_a3, rem_b3;
    reg cout_s1, cout_s2, cout_s3, cout_s4;

    reg [7:0] sum1_d2, sum1_d3, sum1_d4;
    reg [7:0] sum2_d3, sum2_d4;
    reg [7:0] sum3_d4;

    always @(posedge clk) begin
        if (flush) begin
            sum_s1 <= 8'd0; cout_s1 <= 1'b0;
            rem_a1 <= 24'd0; rem_b1 <= 24'd0;

            sum_s2 <= 8'd0; cout_s2 <= 1'b0;
            rem_a2 <= 16'd0; rem_b2 <= 16'd0;

            sum_s3 <= 8'd0; cout_s3 <= 1'b0;
            rem_a3 <= 8'd0; rem_b3 <= 8'd0;

            sum_s4 <= 8'd0; cout_s4 <= 1'b0;

            sum1_d2 <= 8'd0; sum1_d3 <= 8'd0; sum1_d4 <= 8'd0;
            sum2_d3 <= 8'd0; sum2_d4 <= 8'd0;
            sum3_d4 <= 8'd0;

            sum <= 32'd0;
            cout <= 1'b0;
        end
        else begin
            if (!stall) begin
                // stage 1: bit[7:0]
                {cout_s1, sum_s1} <= {1'b0, cin_a[7:0]} + {1'b0, cin_b[7:0]} + cin;
                rem_a1 <= cin_a[31:8];
                rem_b1 <= cin_b[31:8];

                // stage 2: bit[15:8]
                {cout_s2, sum_s2} <= {1'b0, rem_a1[7:0]} + {1'b0, rem_b1[7:0]} + cout_s1;
                rem_a2 <= rem_a1[23:8];
                rem_b2 <= rem_b1[23:8];

                // stage 3: bit[23:16]
                {cout_s3, sum_s3} <= {1'b0, rem_a2[7:0]} + {1'b0, rem_b2[7:0]} + cout_s2;
                rem_a3 <= rem_a2[15:8];
                rem_b3 <= rem_b2[15:8];

                // stage 4: bit[31:24]
                {cout_s4, sum_s4} <= {1'b0, rem_a3} + {1'b0, rem_b3} + cout_s3;

                // shift register alignment
                sum1_d2 <= sum_s1;  sum1_d3 <= sum1_d2;  sum1_d4 <= sum1_d3;
                sum2_d3 <= sum_s2;  sum2_d4 <= sum2_d3;
                sum3_d4 <= sum_s3;

                // output assembly
                sum[7:0]   <= sum1_d4;
                sum[15:8]  <= sum2_d4;
                sum[23:16] <= sum3_d4;
                sum[31:24] <= sum_s4;
                cout <= cout_s4;
            end
        end
    end
endmodule
\end{lstlisting}

\section{流水线全加器测试激励代码 (liushuixian\_4ji\_sim2.v)}

\begin{lstlisting}[language=Verilog]
module liushuixian_4ji_sim2();
    reg [31:0] cin_a, cin_b;
    reg cin, clk, stall, flush;
    wire cout;
    wire [31:0] sum;

    liushuixian_4ji uut (
        .cin_a(cin_a), .cin_b(cin_b), .cin(cin), .clk(clk),
        .stall(stall), .flush(flush), .cout(cout), .sum(sum)
    );

    always #5 clk = ~clk;

    initial begin
        clk = 0; stall = 0; flush = 0; cin = 0;

        // 1-5 cycles
        cin_a = 32'h11110000; cin_b = 32'h22220000;  #10;
        cin_a = 32'h22220000; cin_b = 32'h33330000;  #10;
        cin_a = 32'h33330000; cin_b = 32'h44440000;  #10;
        cin_a = 32'h44440000; cin_b = 32'h55550000;  #10;
        cin_a = 32'h55550000; cin_b = 32'h66660000;  #10;

        // 6-10 cycles
        cin_a = 32'h01110000; cin_b = 32'h02220000;  #10;
        cin_a = 32'h02220000; cin_b = 32'h03330000;  #10;
        cin_a = 32'h03330000; cin_b = 32'h04440000;  #10;
        cin_a = 32'h04440000; cin_b = 32'h05550000;  #10;
        cin_a = 32'h05550000; cin_b = 32'h06660000;  #10;

        // stall at cycle 10 for 2 cycles
        stall = 1;
        #20;
        stall = 0;
        #30;

        // flush at cycle 15
        flush = 1;
        #10;
        flush = 0;

        cin_a = 32'd100;
        cin_b = 32'd200;

        #60;
        $finish;
    end
endmodule
\end{lstlisting}
